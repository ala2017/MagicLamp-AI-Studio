function handleFolderClick(node) {
    // 切换展开状态
    node.expanded = !node.expanded;
    
    if (node.expanded) {
        // 1. 计算当前布局的中心点
        const currentCenter = {
            x: width / 2,
            y: height / 2
        };
        
        // 2. 计算目标中心点（被点击的节点位置）
        const targetCenter = {
            x: node.x,
            y: node.y
        };
        
        // 3. 计算需要移动的偏移量
        const offset = {
            x: currentCenter.x - targetCenter.x,
            y: currentCenter.y - targetCenter.y
        };
        
        // 4. 移动所有节点，保持相对位置关系
        nodes.forEach(n => {
            if (n.id !== node.id) { // 不包括被点击的节点
                n.x += offset.x;
                n.y += offset.y;
            }
        });
        
        // 5. 将点击的节点固定在中心
        node.x = currentCenter.x;
        node.y = currentCenter.y;
        node.fx = node.x;
        node.fy = node.y;
        
        // 6. 为子节点设置径向布局
        const childNodes = nodes.filter(n => {
            return links.some(link => 
                (link.source.id === node.id && link.target.id === n.id) ||
                (link.target.id === node.id && link.source.id === n.id)
            );
        });
        
        // 设置子节点的初始位置
        const angleStep = (2 * Math.PI) / childNodes.length;
        const radius = 150;
        childNodes.forEach((child, i) => {
            const angle = i * angleStep;
            child.x = node.x + radius * Math.cos(angle);
            child.y = node.y + radius * Math.sin(angle);
        });
    } else {
        // 收起时恢复原有布局
        // 移除固定位置
        node.fx = null;
        node.fy = null;
        
        // 让力导向布局重新计算位置
        simulation.alpha(0.3).restart();
    }
    
    // 更新expandedNodes集合
    if (node.expanded) {
        expandedNodes.add(node.id);
    } else {
        expandedNodes.delete(node.id);
    }
    
    // 更新节点样式
    d3.select(this).classed('expanded', node.expanded);
    
    // 更新可视化
    updateVisualization();
}
这样的效果会是：
点击文件夹前：所有节点在以书签栏为中心的布局中
点击文件夹后：
该文件夹移动到显示区域中心
其他所有节点（包括父节点）整体平移，保持它们之间的相对位置关系
子节点以径向方式展开
再次点击时：恢复以书签栏为中心的布局
这样的交互方式：
保持了节点之间的相对位置关系
提供了平滑的视觉过渡
突出了当前关注的节点
维持了整体布局的稳定性
